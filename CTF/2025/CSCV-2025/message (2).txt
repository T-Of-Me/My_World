HOST = '35.197.152.52'

def send(sock, data):
    if isinstance(data, str):
        data = data.encode()
    sock.sendall(data + b'\n')
    time.sleep(0.1)

def recv(sock, timeout=2):
    sock.settimeout(timeout)
    try:
        data = sock.recv(8192)
        return data.decode('utf-8', errors='ignore')
    except:
        return ""
